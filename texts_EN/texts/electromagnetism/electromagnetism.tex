\begin{section}{Electric and Magnetic Fields, Lorentz Force}
$\vec{F}=q(\vec{E}+\vec{v}\times\vec{B})$
\end{section}
\begin{section}{Electromagnetic Field Equations}
$\vec{\nabla}\cdot\vec{E}=\rho/\epsilon_0$, $\vec{\nabla}\cdot\vec{B}=0$, $\vec{\nabla}\times\vec{E}=-\dot{\vec{B}}$, $\vec{\nabla}\times\vec{B}=\mu_0\vec{j}$
\end{section}
\begin{section}{Electrostatics}
$\vec{F}=q\vec{E}$, $\vec{\nabla}\cdot\vec{E}=\rho/\epsilon_0$, $\vec{\nabla}\times\vec{E}=0$
\end{section}
